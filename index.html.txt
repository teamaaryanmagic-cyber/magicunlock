        </select>
        <button onclick="saveSettings()">Save & Close</button>

    <script>
        let clickCount = 0;
        let clickTimer;

        function checkAdminClick() {
            clickCount++;
            clearTimeout(clickTimer);
            clickTimer = setTimeout(() => {
                clickCount = 0;
            }, 500);

            if (clickCount === 3) {
                document.getElementById('admin-panel').style.display = 'flex';
                clickCount = 0;
            }
        }

        function saveSettings() {
            const name = document.getElementById('name-input').value;
            const card = document.getElementById('card-select').value;

            if (name.trim() !== "") {
                document.getElementById('revealed-name').innerText = name.toUpperCase();
            }
            
            const cardElem = document.getElementById('card-value');
            cardElem.innerText = card;
            if (card.includes('♠') || card.includes('♣')) {
                cardElem.classList.add('black');
            } else {
                cardElem.classList.remove('black');
            }

            document.getElementById('admin-panel').style.display = 'none';
        }
    </script>
